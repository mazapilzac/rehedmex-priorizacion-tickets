## Cron `ir.cron` id 131 — "Priorizacion - recalcular antiguedad y score"

```python
now = datetime.datetime.now()
reactivos = [12, 9, 1]
agendados = [10, 23, 24]
scored = reactivos + agendados
tickets = env['helpdesk.ticket'].search([('team_id','in',scored),('stage_id.fold','=',False)])
for t in tickets:
    if not t.create_date:
        continue
    delta = now - t.create_date
    horas = delta.total_seconds() / 3600.0
    dias = delta.days
    if t.team_id.id in reactivos:
        if horas < 1: a = 0
        elif horas < 4: a = 2
        elif horas < 8: a = 4
        elif horas < 24: a = 8
        else: a = 14
    else:
        if dias < 2: a = 0
        elif dias <= 4: a = 3
        elif dias <= 7: a = 6
        else: a = 10
    score = 999 if t.x_override else (t.x_peso_motivo + t.x_impacto + a)
    if t.x_override or score >= 11: prio = '3'
    elif score >= 6: prio = '2'
    elif score >= 3: prio = '1'
    else: prio = '0'
    vals = {}
    if t.x_antiguedad != a: vals['x_antiguedad'] = a
    if t.priority != prio: vals['priority'] = prio
    col = 1 if t.x_override else 0
    if t.color != col: vals['color'] = col
    if vals: t.write(vals)
```
